cmake_minimum_required(VERSION 3.16)
project(qiyu_device_service LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(QIYU_KERNEL_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../qiyu-kernel" CACHE PATH "Canonical Qiyu kernel checkout")
if(NOT EXISTS "${QIYU_KERNEL_DIR}/include/uapi/linux/a081-nordic.h")
  message(FATAL_ERROR "Set QIYU_KERNEL_DIR to the canonical qiyu-kernel checkout")
endif()
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/kernel-uapi/linux")
foreach(header a081-nordic.h ov7251.h qcom-camss-timing.h)
  configure_file("${QIYU_KERNEL_DIR}/include/uapi/linux/${header}" "${CMAKE_CURRENT_BINARY_DIR}/kernel-uapi/linux/${header}" COPYONLY)
endforeach()
add_library(qiyu_device_api INTERFACE)
target_include_directories(qiyu_device_api INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/include" "${CMAKE_CURRENT_BINARY_DIR}/kernel-uapi")
add_library(qiyu_device_abi STATIC src/abi_reference.cpp src/a081_controller_reference.cpp)
set_target_properties(qiyu_device_abi PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_link_libraries(qiyu_device_abi PUBLIC qiyu_device_api)
add_executable(qiyu-nordic-stream src/nordic-stream.cpp)
target_link_libraries(qiyu-nordic-stream PRIVATE qiyu_device_abi pthread)
add_executable(qiyu-controller-radio src/controller-radio.cpp)
target_link_libraries(qiyu-controller-radio PRIVATE qiyu_device_api pthread)
